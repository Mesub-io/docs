```js title="fetch" lines=false
const response = await fetch('https://api.mesub.io/v1/project', {
    headers: {
        Authorization: `Bearer ${process.env.MESUB_API_KEY}`,
    },
});

    const { code, retryable } = await response.json();
    // branch on `code`
}
const data = await response.json();
```
